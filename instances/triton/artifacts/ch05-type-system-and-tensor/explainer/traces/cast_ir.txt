== k_cast: one arith op per cast branch (tracing-period make_ir) ==
arith.truncf count: 1
arith.extf count: 1
arith.fptosi count: 1
arith.sitofp count: 1
arith.fptoui count: 0
arith.uitofp count: 0
--- cast op lines ---
%1 = arith.truncf %0 : f32 to f16 loc(#loc2)
%2 = arith.extf %1 : f16 to f32 loc(#loc3)
%3 = arith.fptosi %0 : f32 to i32 loc(#loc4)
%4 = arith.sitofp %3 : i32 to f32 loc(#loc5)

== k_bf16_twohop: fp16->bf16 detours via fp32 (two IR ops) ==
arith.extf count: 1
arith.truncf count: 1
%1 = arith.extf %0 : f16 to f32 loc(#loc2)
%2 = arith.truncf %1 : f32 to bf16 loc(#loc2)
